function calculateBenchmark() {
  const arr = document.getElementById('arr-range').value;
  const employees = parseInt(document.getElementById('employee-count').value);
  const techSpend = parseInt(document.getElementById('tech-spend').value);
  const challenge = document.getElementById('primary-challenge').value;
  
  // Simple benchmark calculations
  const arrMidpoint = {
    '5-10': 7500000,
    '10-20': 15000000,
    '20-40': 30000000,
    '40-60': 50000000,
    '60+': 80000000
  };
  
  const expectedSpend = arrMidpoint[arr] * 0.08; // 8% of ARR benchmark
  const spendDifference = techSpend - expectedSpend;
  const spendPercentage = (techSpend / arrMidpoint[arr]) * 100;
  
  let resultHTML = `
    <div class="benchmark-metric">
      <strong>Your Tech Spend as % of ARR:</strong> ${spendPercentage.toFixed(1)}%<br>
      <strong>Industry Benchmark:</strong> 7-9% of ARR<br>
      <strong>Potential Annual Savings:</strong> $${Math.max(0, spendDifference).toLocaleString()}
    </div>
  `;
  
  if (spendPercentage > 10) {
    resultHTML += `
      <div class="notice--warning">
        <strong>⚠️ Above Benchmark:</strong> Your technology spend is higher than typical for your revenue range. 
        Most companies your size save $${(spendDifference/1000000).toFixed(1)}M annually through optimization.
      </div>
    `;
  } else if (spendPercentage < 6) {
    resultHTML += `
      <div class="notice--info">
        <strong>ℹ️ Below Benchmark:</strong> Your technology spend is lean. 
        Focus on strategic investments in automation and scaling infrastructure.
      </div>
    `;
  } else {
    resultHTML += `
      <div class="notice--success">
        <strong>✓ Within Range:</strong> Your spending aligns with benchmarks. 
        Optimization opportunities likely exist in specific areas rather than overall spend.
      </div>
    `;
  }
  
  document.getElementById('results-content').innerHTML = resultHTML;
  document.getElementById('benchmark-results').classList.remove('hidden');
  
  // Scroll to results
  document.getElementById('benchmark-results').scrollIntoView({ behavior: 'smooth' });
}
</script>
